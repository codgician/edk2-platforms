# TestSupport.c builds the production client, allocator, RPC dispatcher and filesystem together.
function(optee_host_options NAME)
  target_link_options(${NAME} PRIVATE
    -Wl,--wrap=AllocatePool -Wl,--wrap=AllocateZeroPool
    -Wl,--wrap=ReallocatePool -Wl,--wrap=FreePool)
endfunction()
sky1_host_test(optee_client PRELUDE TestConfig.h SOURCES ClientTest.c TestSupport.c)
optee_host_options(optee_client)
foreach(CASE zero ownership size metadata close open open_rejected partial rpc loop transport)
  add_test(NAME optee_client_${CASE} COMMAND optee_client ${CASE})
endforeach()
sky1_host_test(reef_fs_test PRELUDE TestConfig.h SOURCES ReeFsTest.c TestSupport.c)
optee_host_options(reef_fs_test)
foreach(CASE stale huge direction sequence)
  add_test(NAME reef_fs_${CASE} COMMAND reef_fs_test ${CASE})
endforeach()
