# Compile the production callback implementation; discard unrelated HII display paths.
file(READ "${CMAKE_CURRENT_LIST_DIR}/../Tcg2ConfigStrings.uni" SETUP_STRINGS)
string(REGEX MATCHALL "#string[ \t]+[A-Za-z0-9_]+" SETUP_TOKENS "${SETUP_STRINGS}")
set(SETUP_TOKEN_HEADER "")
foreach(TOKEN IN LISTS SETUP_TOKENS)
  string(REGEX REPLACE "#string[ \t]+" "" TOKEN "${TOKEN}")
  string(APPEND SETUP_TOKEN_HEADER "#define ${TOKEN} 1\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SetupTokens.h" "${SETUP_TOKEN_HEADER}")
